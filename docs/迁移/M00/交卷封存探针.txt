function send(path, method = "GET", cookie, body) {
  return app.request(path, {
    method,
    headers: { ...(cookie ? { cookie } : {}), ...(typeof body === "string" ? { "content-type": "application/json" } : {}) },
    body,
  });
}

function submitBody(revisions, receipt) {
  const evidence = revisions.map(({ questionId: id }) => id === questionId
    ? { questionId: id, state: "frozen", versionId: receipt.versionId, revision: receipt.revision }
    : { questionId: id, state: "none" });
  return JSON.stringify({ revisions, evidence });
}
function facts(conn) {
  return {
    attempt: select(conn, "SELECT status, submitted_at, score_auto FROM attempts WHERE id = ?", attemptId),
    responses: select(conn, "SELECT id, question_id, answer_revision, auto_correct, final_correct FROM responses WHERE attempt_id = ? ORDER BY question_id", attemptId),
    evidence: select(conn, "SELECT id, question_id, state, version_id, recorded_at FROM submission_evidence WHERE attempt_id = ? ORDER BY question_id", attemptId),
    markers: select(conn, "SELECT payload_json FROM events WHERE attempt_id = ? AND type = 'submit'", attemptId),
    noteHead: select(conn, "SELECT id, current_revision, current_version_id FROM notes WHERE attempt_id = ? AND question_id = ? AND phase = 'scratch'", attemptId, questionId),
    versions: select(conn, "SELECT v.id, v.revision, v.hash, v.stroke_count, v.point_count FROM note_versions v JOIN notes n ON n.id = v.note_id WHERE n.attempt_id = ? AND n.question_id = ? ORDER BY v.revision", attemptId, questionId),
  };
}

function resultProjection(body) {
  const { data } = attemptResultOkSchema.parse(body);
  assert.equal(data.answersReleased, false);
  assert.equal(data.dueAt, dueAt);
  assert.equal(data.attempt.scoreAuto, null);
  assert.equal(data.summary.scoreFinal, null);
  assert.equal(data.summary.pendingCount, null);
  const rows = data.units.flatMap((unit) => unit.questions);
  assert.equal(rows.length, 8);
  for (const row of rows) {
    for (const field of ["answers", "solutionMd", "autoCorrect", "finalCorrect", "teacherMark", "teacherComment"]) assert.equal(row[field], null);
  }
  assert.deepEqual(rows.find((row) => row.questionId === questionId)?.answer, finalAnswer);
  assert.doesNotMatch(JSON.stringify(body), /故选 B|\[\[(?:正确|4)\]\]|"(?:hints|sourceMd|bodyPath)"\s*:/);
  return { answersReleased: data.answersReleased, dueAt: data.dueAt, status: data.attempt.status, scoreAuto: data.attempt.scoreAuto, summary: data.summary, questionCount: rows.length, publicQuestion: rows.find((row) => row.questionId === questionId) };
}
